│ Context                                                                                                                                                                                                                                                  │
│                                                                                                                                                                                                                                                          │
│ Tk is a cross-platform GUI toolkit (~180K lines of C, ~250KB of Tcl scripts) tightly coupled to the Tcl interpreter. This plan ports it to pure Go as module takigo, eliminating all Tcl/Tk runtime dependencies. The Go API uses functional options     │
│ (tk.NewButton(parent, tk.Text("Hello"), tk.Command(fn))). The initial platform target is X11/Linux via cgo Xlib bindings. The C sources in tk/ are kept as reference.                                                                                    │

│ Event Loop Architecture                                                                                                                                                                                                                                  │
│                                                                                                                                                                                                                                                          │
│ type App struct {                                                                                                                                                                                                                                        │
│     display   *xlib.Display                                                                                                                                                                                                                              │
│     eventCh   chan xlib.Event       // raw X events from reader goroutine                                                                                                                                                                                │
│     idleCh    chan func()            // idle callbacks (replaces Tcl_DoWhenIdle)                                                                                                                                                                         │
│     timerCh   chan timerEntry        // scheduled callbacks                                                                                                                                                                                              │
│     mainCh    chan func()            // cross-goroutine calls via RunOnMain                                                                                                                                                                              │
│     done      chan struct{}                                                                                                                                                                                                                              │
│     windows   map[xlib.Window]*window.Window                                                                                                                                                                                                             │
│ }                                                                                                                                                                                                                                                        │
│                                                                                                                                                                                                                                                          │
│ - A dedicated goroutine calls XNextEvent in a loop, posting to eventCh                                                                                                                                                                                   │
│ - All widget operations happen on the main goroutine (single-thread model)                                                                                                                                                                               │
│ - app.RunOnMain(fn) provides safe cross-goroutine access                                                                                                                                                                                                 │
│ - Timer events use time.Timer feeding into the select loop                                                                                                                                                                                               │
│ - Idle callbacks coalesce redraw/layout requests                                                                                                                                                                                                         │
│                                                                                                                                                                                                                                                          │

│ Tcl→Go Translation Patterns                                                                                                                                                                                                                              │
│                                                                                                                                                                                                                                                          │
│ ┌──────────────────────────────┬────────────────────────────────────────────────────────────┐                                                                                                                                                            │
│ │         Tcl Pattern          │                       Go Equivalent                        │                                                                                                                                                            │
│ ├──────────────────────────────┼────────────────────────────────────────────────────────────┤                                                                                                                                                            │
│ │ proc ::tk::ButtonDown {w}    │ func buttonDown(b *Button)                                 │                                                                                                                                                            │
│ ├──────────────────────────────┼────────────────────────────────────────────────────────────┤                                                                                                                                                            │
│ │ bind Button <Enter> {script} │ Register handler in widget init() as class binding         │                                                                                                                                                            │
│ ├──────────────────────────────┼────────────────────────────────────────────────────────────┤                                                                                                                                                            │
│ │ $w configure -relief sunken  │ b.SetRelief(ReliefSunken)                                  │                                                                                                                                                            │
│ ├──────────────────────────────┼────────────────────────────────────────────────────────────┤                                                                                                                                                            │
│ │ after 500 [list $w invoke]   │ time.AfterFunc(500*ms, func() { app.RunOnMain(b.Invoke) }) │                                                                                                                                                            │
│ ├──────────────────────────────┼────────────────────────────────────────────────────────────┤                                                                                                                                                            │
│ │ set ::tk::Priv(var) val      │ Package-level or App-level state struct                    │                                                                                                                                                            │
│ ├──────────────────────────────┼────────────────────────────────────────────────────────────┤                                                                                                                                                            │
